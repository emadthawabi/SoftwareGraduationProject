# Example: one table (العملاء) end to end

    pip install -r requirements.txt
    cp .env.example .env            # then edit the values
    alembic upgrade head            # creates the tables in PostgreSQL
    uvicorn app.main:app --reload   # API docs at http://127.0.0.1:8000/docs
    pytest                          # runs the tests (SQLite, no PostgreSQL needed)

Layout
    app/core/config.py     settings from environment
    app/core/security.py   reads the login cookie, returns who is calling and for which office
    app/db/base.py         Base class and the standard fields every table has
    app/db/session.py      one database session per request, told which office it serves
    app/models/            the tables (SQLAlchemy)
    app/schemas/           what the API accepts and returns (Pydantic)
    app/api/               the endpoints
    migrations/            Alembic: the history of schema changes
    tests/                 automated tests

import uuid

from fastapi import APIRouter, Depends, HTTPException, Query, Response, status
from sqlalchemy import func, select
from sqlalchemy.orm import Session

from app.core.security import AuthContext, get_auth
from app.db.session import get_db
from app.models.client import Client
from app.schemas.client import ClientCreate, ClientOut, ClientUpdate

router = APIRouter(prefix="/clients", tags=["العملاء"])


def _get_or_404(db: Session, auth: AuthContext, client_id: uuid.UUID) -> Client:
    client = db.scalar(
        select(Client).where(
            Client.id == client_id,
            Client.office_id == auth.office_id,   # office comes from the token
            Client.deleted_at.is_(None),          # hide records marked as deleted
        )
    )
    if client is None:
        raise HTTPException(status.HTTP_404_NOT_FOUND, "العميل غير موجود")
    return client


@router.get("", response_model=list[ClientOut])
def list_clients(
    q: str | None = Query(default=None, description="بحث بالاسم"),
    limit: int = Query(default=50, le=200),
    offset: int = 0,
    db: Session = Depends(get_db),
    auth: AuthContext = Depends(get_auth),
):
    stmt = select(Client).where(Client.office_id == auth.office_id, Client.deleted_at.is_(None))
    if q:
        stmt = stmt.where(Client.name.ilike(f"%{q}%"))
    return db.scalars(stmt.order_by(Client.name).limit(limit).offset(offset)).all()


@router.post("", response_model=ClientOut, status_code=status.HTTP_201_CREATED)
def create_client(body: ClientCreate, db: Session = Depends(get_db), auth: AuthContext = Depends(get_auth)):
    client = Client(**body.model_dump(), office_id=auth.office_id, created_by=auth.user_id)
    db.add(client)
    db.commit()
    return client


@router.get("/{client_id}", response_model=ClientOut)
def get_client(client_id: uuid.UUID, db: Session = Depends(get_db), auth: AuthContext = Depends(get_auth)):
    return _get_or_404(db, auth, client_id)


@router.patch("/{client_id}", response_model=ClientOut)
def update_client(
    client_id: uuid.UUID, body: ClientUpdate, db: Session = Depends(get_db), auth: AuthContext = Depends(get_auth)
):
    client = _get_or_404(db, auth, client_id)
    for field, value in body.model_dump(exclude_unset=True).items():
        setattr(client, field, value)
    client.updated_by = auth.user_id
    db.commit()
    return client


@router.delete("/{client_id}", status_code=status.HTTP_204_NO_CONTENT)
def delete_client(client_id: uuid.UUID, db: Session = Depends(get_db), auth: AuthContext = Depends(get_auth)):
    client = _get_or_404(db, auth, client_id)
    client.deleted_at = func.now()        # marked as deleted, never erased
    client.updated_by = auth.user_id
    db.commit()
    return Response(status_code=status.HTTP_204_NO_CONTENT)

===== FILE: app/core/security.py =====
import uuid
from dataclasses import dataclass

import jwt
from fastapi import Cookie, HTTPException, status

from app.core.config import settings


@dataclass(frozen=True)
class AuthContext:
    """Who is calling. Always taken from the token, never from the request body."""
    user_id: uuid.UUID
    office_id: uuid.UUID
    role: str


def get_auth(access_token: str | None = Cookie(default=None)) -> AuthContext:
    if not access_token:
        raise HTTPException(status.HTTP_401_UNAUTHORIZED, "غير مسجّل الدخول")
    try:
        data = jwt.decode(access_token, settings.jwt_secret, algorithms=[settings.jwt_algorithm])
        return AuthContext(uuid.UUID(data["sub"]), uuid.UUID(data["office"]), data["role"])
    except (jwt.PyJWTError, KeyError, ValueError):
        raise HTTPException(status.HTTP_401_UNAUTHORIZED, "جلسة غير صالحة أو منتهية")

import uuid
from datetime import datetime

from sqlalchemy import DateTime, ForeignKey, func
from sqlalchemy.orm import DeclarativeBase, Mapped, declared_attr, mapped_column


class Base(DeclarativeBase):
    pass


class StandardMixin:
    """The standard fields every table carries."""

    id: Mapped[uuid.UUID] = mapped_column(primary_key=True, default=uuid.uuid4)
    created_at: Mapped[datetime] = mapped_column(DateTime(timezone=True), server_default=func.now())
    updated_at: Mapped[datetime] = mapped_column(
        DateTime(timezone=True), server_default=func.now(), onupdate=func.now()
    )
    deleted_at: Mapped[datetime | None] = mapped_column(DateTime(timezone=True))  # "mark as deleted"

    # Columns with a ForeignKey on a mixin must be declared with declared_attr.
    @declared_attr
    def office_id(cls) -> Mapped[uuid.UUID]:
        return mapped_column(ForeignKey("offices.id"), index=True)

    @declared_attr
    def created_by(cls) -> Mapped[uuid.UUID | None]:
        return mapped_column(ForeignKey("users.id"))

    @declared_attr
    def updated_by(cls) -> Mapped[uuid.UUID | None]:
        return mapped_column(ForeignKey("users.id"))

from collections.abc import Iterator

from fastapi import Depends
from sqlalchemy import create_engine, text
from sqlalchemy.orm import Session, sessionmaker

from app.core.config import settings
from app.core.security import AuthContext, get_auth

engine = create_engine(settings.database_url, pool_pre_ping=True)
SessionLocal = sessionmaker(engine, expire_on_commit=False)


def get_db(auth: AuthContext = Depends(get_auth)) -> Iterator[Session]:
    """One database session per request, already told which office it belongs to."""
    with SessionLocal() as db:
        if db.get_bind().dialect.name == "postgresql":
            # PostgreSQL row-level security reads this value, so even a query that
            # forgets its office filter cannot return another office's rows.
            db.execute(text("SELECT set_config('app.office_id', :o, true)"), {"o": str(auth.office_id)})
        yield db

import uuid

from pydantic import BaseModel, ConfigDict, Field

from app.models.client import ClientType


class ClientBase(BaseModel):
    type: ClientType
    name: str = Field(min_length=1, max_length=200)
    national_id: str | None = None
    phone: str | None = None
    email: str | None = None
    address: str | None = None
    notes: str | None = None


class ClientCreate(ClientBase):
    """What the API accepts when creating. No id, no office: the server sets them."""

    """Every field optional, so PATCH changes only what is sent."""
    type: ClientType | None = None
    name: str | None = Field(default=None, min_length=1, max_length=200)
    national_id: str | None = None
    phone: str | None = None

def run_migrations_offline() -> None:
    """Run migrations in 'offline' mode.

    This configures the context with just a URL
    and not an Engine, though an Engine is acceptable
    here as well.  By skipping the Engine creation
    we don't even need a DBAPI to be available.

    Calls to context.execute() here emit the given string to the
    script output.

    """
    url = config.get_main_option("sqlalchemy.url")
    context.configure(
        url=url,
        target_metadata=target_metadata,
        literal_binds=True,
        dialect_opts={"paramstyle": "named"},
    )

    with context.begin_transaction():
        context.run_migrations()


def run_migrations_online() -> None:
    """Run migrations in 'online' mode.

    In this scenario we need to create an Engine
    and associate a connection with the context.

    """
    connectable = engine_from_config(
        config.get_section(config.config_ini_section, {}),
        prefix="sqlalchemy.",
        poolclass=pool.NullPool,
    )

    with connectable.connect() as connection:
        context.configure(
            connection=connection, target_metadata=target_metadata
        )

        with context.begin_transaction():
            context.run_migrations()


if context.is_offline_mode():
    run_migrations_offline()
else:
    run_migrations_online()

def token(office, user, role="lawyer"):
    return jwt.encode({"sub": str(user), "office": str(office), "role": role}, settings.jwt_secret, settings.jwt_algorithm)


def test_clients_flow():
    Base.metadata.drop_all(engine); Base.metadata.create_all(engine)
    o1, o2 = uuid.uuid4(), uuid.uuid4(); u1, u2 = uuid.uuid4(), uuid.uuid4()
